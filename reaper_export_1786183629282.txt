local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera or workspace:WaitForChild("Camera")
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui", 10)

-- Check Drawing API
local DrawingLib = (type(Drawing) ~= "nil" and Drawing.new)

--------------------------------------------------
-- 1. SETTINGS & STATES
--------------------------------------------------
local States = {
    Aimbot = false, LockLevel = 1, TargetPart = "Head", UseFOV = true, FOVRadius = 150,
    HitboxEnabled = false, HitboxSize = 10, HitboxTransparency = 0.7,
    ESP_Highlight = false, ESP_Tracers = false, ESP_NameTags = false
}
local SmoothnessMap = {[1] = 0.15, [2] = 0.40, [3] = 1.00}
local ESP_Players = {}

--------------------------------------------------
-- 2. UI CONSTRUCTION (สร้างก่อนเพื่อให้รันติดแน่นอน)
--------------------------------------------------
local gui = Instance.new("ScreenGui", PlayerGui)
gui.Name = "ReaperProject_Final"
gui.ResetOnSpawn = false

local main = Instance.new("Frame", gui)
main.Size = UDim2.fromOffset(620, 440)
main.Position = UDim2.new(0.5, -310, 0.5, -220)
main.BackgroundColor3 = Color3.fromRGB(15, 15, 18)
main.BorderSizePixel = 0
main.Active = true
main.Draggable = true -- ถ้าลากไม่ได้ให้เปลี่ยนเป็นสคริปต์ลากมือ

local mainCorner = Instance.new("UICorner", main)
mainCorner.CornerRadius = UDim.new(0, 15)

local sidebar = Instance.new("Frame", main)
sidebar.Size = UDim2.new(0, 140, 1, -60)
sidebar.Position = UDim2.fromOffset(15, 50)
sidebar.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
Instance.new("UICorner", sidebar).CornerRadius = UDim.new(0, 10)

content.Size = UDim2.new(1, -180, 1, -60)
content.Position = UDim2.fromOffset(165, 50)
content.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
Instance.new("UICorner", content).CornerRadius = UDim.new(0, 10)

-- FOV Circle
local fovCircle = Instance.new("Frame", gui)
fovCircle.Size = UDim2.fromOffset(States.FOVRadius*2, States.FOVRadius*2)

fovCircle.BackgroundTransparency = 1
fovCircle.Visible = false

--------------------------------------------------
-- 3. LOGIC FUNCTIONS
--------------------------------------------------
local function GetClosestTarget()
    local closest, shortestDist = nil, math.huge
    local mousePos = UserInputService:GetMouseLocation()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild(States.TargetPart) then
            local pos, onScreen = Camera:WorldToViewportPoint(p.Character[States.TargetPart].Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                if dist < (States.UseFOV and States.FOVRadius or math.huge) and dist < shortestDist then
                    shortestDist, closest = dist, p
                end
            end
        end
    end
    return closest
end

local function UpdateHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                if States.HitboxEnabled then
                    root.Size = Vector3.new(States.HitboxSize, States.HitboxSize, States.HitboxSize)
                    root.Transparency = States.HitboxTransparency
                    root.Color = Color3.fromRGB(255, 0, 0)
                    root.CanCollide = false
                else
                    root.Size = Vector3.new(2, 2, 1)
                    root.Transparency = 1
                end
            end
        end
    end
end

local function CreateESP(p)
    if not DrawingLib or p == LocalPlayer or ESP_Players[p] then return end
    pcall(function()
        ESP_Players[p] = {
            Tracer = Drawing.new("Line"),
            Name = Drawing.new("Text")
        }
        local obj = ESP_Players[p]
        obj.Tracer.Visible = false; obj.Tracer.Thickness = 1; obj.Tracer.Color = Color3.new(1,0,0)
        obj.Name.Visible = false; obj.Name.Size = 14; obj.Name.Center = true; obj.Name.Outline = true
    end)
end

local function UpdateESP()
    for p, obj in pairs(ESP_Players) do
        if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local pos, onScreen = Camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
            if onScreen then
                if States.ESP_Tracers then
                    obj.Tracer.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
                    obj.Tracer.To = Vector2.new(pos.X, pos.Y); obj.Tracer.Visible = true
                else obj.Tracer.Visible = false end
                if States.ESP_NameTags then
                    obj.Name.Position = Vector2.new(pos.X, pos.Y - 30); obj.Name.Text = p.Name; obj.Name.Visible = true
                else obj.Name.Visible = false end
            else obj.Tracer.Visible = false; obj.Name.Visible = false end
        else obj.Tracer.Visible = false; obj.Name.Visible = false end
    end
end

--------------------------------------------------
-- 4. UI BUILDER (Tabs & Toggles)

local panels = {}
local function createTab(name, id)
    local p = Instance.new("ScrollingFrame", content)
    p.Size = UDim2.new(1, -10, 1, -10)
    p.BackgroundTransparency = 1
    p.Visible = false
    Instance.new("UIListLayout", p).Padding = UDim.new(0, 5)
    panels[id] = p
    
    local b = Instance.new("TextButton", sidebar)
    b.Size = UDim2.new(1, 0, 0, 35)
    b.Text = name
    b.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    b.TextColor3 = Color3.white
    Instance.new("UICorner", b)
    b.MouseButton1Click:Connect(function()
        for _, v in pairs(panels) do v.Visible = false end
        p.Visible = true
    end)
end

local function createToggle(panel, text, callback)
    local state = false
    local b = Instance.new("TextButton", panel)
    b.Size = UDim2.new(1, 0, 0, 35)
    b.Text = text .. " : OFF"
    b.BackgroundColor3 = Color3.fromRGB(40, 40, 45)
    b.TextColor3 = Color3.white
    Instance.new("UICorner", b)
    b.MouseButton1Click:Connect(function()
        state = not state
        b.Text = text .. " : " .. (state and "ON" or "OFF")
        b.BackgroundColor3 = state and Color3.fromRGB(200, 50, 50) or Color3.fromRGB(40, 40, 45)
        callback(state)
    end)
end

createTab("🎯 AIMBOT", "aim")
createTab("📦 HITBOX", "hb")
createTab("👁 VISUALS", "vis")
panels.aim.Visible = true

createToggle(panels.aim, "Aimbot", function(s) States.Aimbot = s fovCircle.Visible = s end)
createToggle(panels.hb, "Hitbox", function(s) States.HitboxEnabled = s end)
createToggle(panels.vis, "Tracers", function(s) States.ESP_Tracers = s end)

--------------------------------------------------
-- 5. LOOPS & SETUP
--------------------------------------------------
RunService.RenderStepped:Connect(function()
    if States.Aimbot then
        local t = GetClosestTarget()
        if t and t.Character then
            Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, t.Character[States.TargetPart].Position), SmoothnessMap[States.LockLevel])
        end
    end
    if DrawingLib then pcall(UpdateESP) end
    if fovCircle.Visible then
        local m = UserInputService:GetMouseLocation()
        fovCircle.Position = UDim2.fromOffset(m.X, m.Y)
    end
end)

RunService.Heartbeat:Connect(function()
    pcall(UpdateHitboxes)
end)

local toggleBtn = Instance.new("TextButton", gui)
toggleBtn.Size = UDim2.fromOffset(40, 40)
toggleBtn.Position = UDim2.new(0, 10, 0.5, 0)
toggleBtn.Text = "MENU"
toggleBtn.MouseButton1Click:Connect(function() main.Visible = not main.Visible end)

for _, p in ipairs(Players:GetPlayers()) do CreateESP(p) end
Players.PlayerAdded:Connect(CreateESP)
