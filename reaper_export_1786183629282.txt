	Aimbot = false, LockLevel = 1, TargetPart = "Head", UseFOV = true, FOVRadius = 150,
	HitboxEnabled = false, HitboxSize = 10, HitboxTransparency = 0.7,
	ESP_Highlight = false, ESP_Tracers = false, ESP_NameTags = false,

-- 2. CORE LOGIC FUNCTIONS

local function GetClosestTarget()
	local closestPlayer = nil
	local shortestDistance = math.huge
	local mousePos = UserInputService:GetMouseLocation()

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild(States.TargetPart) then
			local pos, onScreen = Camera:WorldToViewportPoint(player.Character[States.TargetPart].Position)
			if onScreen then
				local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
				if dist < (States.UseFOV and States.FOVRadius or math.huge) and dist < shortestDistance then
					shortestDistance = dist
					closestPlayer = player
				end
			end
		end
	end
	return closestPlayer
end

local function UpdateHitboxes()
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			local root = player.Character:FindFirstChild("HumanoidRootPart")
			if root then
				if States.HitboxEnabled then
					root.Size = Vector3.new(States.HitboxSize, States.HitboxSize, States.HitboxSize)
					root.Transparency = States.HitboxTransparency
					root.Color = Color3.fromRGB(255, 0, 0)
					root.CanCollide = false
				else
					root.Size = Vector3.new(2, 2, 1)
					root.Transparency = 1
					root.CanCollide = true
				end
			end
		end
	end
end

local function CreateESP(player)
	if not DrawingLib or player == LocalPlayer or ESP_Players[player] then return end
	local objects = {
		Tracer = Drawing.new("Line"),
		Name = Drawing.new("Text"),
		Health = Drawing.new("Text")
	}
	objects.Tracer.Thickness = 1
	objects.Tracer.Color = Color3.fromRGB(255, 50, 50)
	objects.Name.Size = 14; objects.Name.Center = true; objects.Name.Outline = true
	objects.Health.Size = 13; objects.Health.Center = true; objects.Health.Outline = true
	ESP_Players[player] = objects
end

local function UpdateESP()
	for player, objects in pairs(ESP_Players) do
		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChild("Humanoid")
		if root and hum and hum.Health > 0 then
			local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
			if onScreen then
				if States.ESP_Tracers then
					objects.Tracer.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
					objects.Tracer.To = Vector2.new(pos.X, pos.Y); objects.Tracer.Visible = true
				else objects.Tracer.Visible = false end
				if States.ESP_NameTags then
					local headPos = Camera:WorldToViewportPoint(root.Position + Vector3.new(0, 3, 0))
					objects.Name.Position = Vector2.new(headPos.X, headPos.Y); objects.Name.Text = player.Name; objects.Name.Visible = true
					objects.Health.Position = Vector2.new(headPos.X, headPos.Y + 15); objects.Health.Text = "HP: "..math.floor(hum.Health); objects.Health.Visible = true
				else objects.Name.Visible = false; objects.Health.Visible = false end
			else for _,o in pairs(objects) do o.Visible = false end end
		else for _,o in pairs(objects) do o.Visible = false end end
	end
end

local function createToggle(panel, text, default, callback)
	local state = default
	local btn = Instance.new("TextButton", panel); btn.Size = UDim2.new(1, -8, 0, 34); btn.BackgroundColor3 = state and Color3.fromRGB(220, 40, 40) or Color3.fromRGB(32, 32, 38); btn.Text = text.." : "..(state and "ON" or "OFF"); btn.TextColor3 = Color3.fromRGB(240, 240, 240); btn.Font = Enum.Font.Code
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
	btn.MouseButton1Click:Connect(function()
		state = not state
		btn.BackgroundColor3 = state and Color3.fromRGB(220, 40, 40) or Color3.fromRGB(32, 32, 38)
		btn.Text = text.." : "..(state and "ON" or "OFF")
		callback(state)
	end)
end

	if not target or not LocalPlayer.Character then return end
	local bullet = Instance.new("Part", workspace); bullet.Size = Vector3.new(1,1,3); bullet.Color = Color3.new(1,0,0); bullet.Material = Enum.Material.Neon; bullet.Anchored = true; bullet.CanCollide = false
	bullet.CFrame = CFrame.new(LocalPlayer.Character.HumanoidRootPart.Position, target.Character.HumanoidRootPart.Position)

		for i=1, 200 do
			if not bullet.Parent or not target.Character then break end
