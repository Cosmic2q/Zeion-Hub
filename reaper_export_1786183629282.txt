	btn.BackgroundColor3 = state and Color3.fromRGB(220, 40, 40) or Color3.fromRGB(32, 32, 38)

	return btn

-- 2. PROJECTILE
local function GetClosestTarget()
	local closestPlayer = nil
	local shortestDistance = math.huge
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			local dist = (player.Character.HumanoidRootPart.Position - Camera.CFrame.Position).Magnitude
			if dist < shortestDistance then
				shortestDistance = dist
				closestPlayer = player
			end
		end
	end
	return closestPlayer
end

createButton(panels.Projectile, "🔥 Fire Homing Bullet", function()
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return end
	local targetPlayer = GetClosestTarget()
	if not targetPlayer or not targetPlayer.Character or not targetPlayer.Character:FindFirstChild("HumanoidRootPart") then return end

	local bullet = Instance.new("Part")
	bullet.Size = Vector3.new(1, 1, 3)
	bullet.Color = Color3.fromRGB(255, 50, 50)
	bullet.Material = Enum.Material.Neon
	bullet.CanCollide = false
	bullet.Anchored = true
	bullet.CFrame = CFrame.new(char.HumanoidRootPart.Position + Vector3.new(0, 2, 0), targetPlayer.Character.HumanoidRootPart.Position)
	bullet.Parent = workspace

	task.spawn(function()
		local lifetime = 0
		while bullet and bullet.Parent and lifetime < 5 do
			local dt = RunService.Heartbeat:Wait()
			lifetime = lifetime + dt
			if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
				local targetPart = targetPlayer.Character.HumanoidRootPart
				bullet.CFrame = bullet.CFrame:Lerp(CFrame.new(bullet.Position, targetPart.Position), States.HomingTurnRate) * CFrame.new(0, 0, -States.HomingSpeed * dt)
				if (bullet.Position - targetPart.Position).Magnitude < 3 then
					bullet:Destroy()
					break
				end
			else
				bullet.CFrame = bullet.CFrame * CFrame.new(0, 0, -States.HomingSpeed * dt)
			end
		end
		if bullet and bullet.Parent then bullet:Destroy() end
	end)
end)

-- 3. HITBOX (Optimized & Forced Version)
--------------------------------------------------
local function UpdateHitboxes()
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			local root = player.Character:FindFirstChild("HumanoidRootPart")
			if root then
				if States.HitboxEnabled then
					-- เช็คก่อนเปลี่ยนค่าเพื่อประหยัด CPU (Optimization)
					if root.Size.X ~= States.HitboxSize then
						root.Size = Vector3.new(States.HitboxSize, States.HitboxSize, States.HitboxSize)
						root.Transparency = States.HitboxTransparency
						root.Material = Enum.Material.ForceField
						root.Color = Color3.fromRGB(255, 0, 0)
						root.CanCollide = false -- กันเราเดินชนแล้วกระเด็น
					end
				else
					-- คืนค่าปกติ
					if root.Size.X ~= 2 then
						root.Size = Vector3.new(2, 2, 1)
						root.Transparency = 1
						root.Material = Enum.Material.Plastic
						root.CanCollide = true
					end
				end
			end
		end
	end
end

createToggle(panels.Hitbox, "Hitbox Expander", false, function(state)
	States.HitboxEnabled = state
	if not state then UpdateHitboxes() end -- รีเซ็ตทันทีที่ปิด
end)

local ESP_Players = {} -- ** แก้จุดที่หายไป **

local function CreateESP(player)
	if not DrawingLib or player == LocalPlayer or ESP_Players[player] then return end

	local objects = {
		Tracer = Drawing.new("Line"),
		Name = Drawing.new("Text"),
		Health = Drawing.new("Text")
	}
	objects.Tracer.Thickness = 1
	objects.Tracer.Color = Color3.fromRGB(255, 50, 50)
	objects.Name.Size = 14
	objects.Name.Center = true
	objects.Name.Outline = true
	objects.Name.Color = Color3.fromRGB(255, 255, 255)
	objects.Health.Size = 13
	objects.Health.Center = true
	objects.Health.Outline = true
	ESP_Players[player] = objects
end

local function RemoveESP(player)
	if ESP_Players[player] then
		for _, obj in pairs(ESP_Players[player]) do obj:Remove() end
		ESP_Players[player] = nil
	end
end

local function UpdateESP()
	for player, objects in pairs(ESP_Players) do
		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChild("Humanoid")
		if root and hum and hum.Health > 0 then
			local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
			if onScreen then
				if States.ESP_Tracers then
					objects.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
					objects.Tracer.To = Vector2.new(pos.X, pos.Y)
					objects.Tracer.Visible = true
				else objects.Tracer.Visible = false end

				if States.ESP_NameTags then
					local headPos = Camera:WorldToViewportPoint(root.Position + Vector3.new(0, 3, 0))
					objects.Name.Position = Vector2.new(headPos.X, headPos.Y)
					objects.Name.Text = player.Name
					objects.Name.Visible = true
					objects.Health.Position = Vector2.new(headPos.X, headPos.Y + 15)
					objects.Health.Text = "HP: " .. math.floor(hum.Health)
					objects.Health.Color = Color3.fromHSV(math.clamp(hum.Health/100, 0, 1) * 0.3, 1, 1)
					objects.Health.Visible = true
				else objects.Name.Visible = false objects.Health.Visible = false end
			else objects.Tracer.Visible = false objects.Name.Visible = false objects.Health.Visible = false end
		else objects.Tracer.Visible = false objects.Name.Visible = false objects.Health.Visible = false end
	end
end

Players.PlayerAdded:Connect(CreateESP)
Players.PlayerRemoving:Connect(RemoveESP)
for _, p in ipairs(Players:GetPlayers()) do CreateESP(p) end
